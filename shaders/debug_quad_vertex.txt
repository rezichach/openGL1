#version 330 core

// Vertex attributes
layout (location = 0) in vec3 aPos;       // Vertex position (already in NDC space)
layout (location = 1) in vec2 aTexCoords; // Texture coordinates

// Output to fragment shader
out vec2 TexCoords;

void main()
{
    // Pass texture coordinates to fragment shader
    TexCoords = aTexCoords;
    
    // No transformations needed - the quad vertices are already in normalized device coordinates (-1 to 1)
    // This is used for rendering a full-screen quad to display the shadow map
    gl_Position = vec4(aPos, 1.0);
}
